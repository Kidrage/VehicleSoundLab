cmake_minimum_required(VERSION 3.21)
project(VslFrozenTextureProbe LANGUAGES C)
set(CMAKE_C_STANDARD 11)
get_filename_component(VSL_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(texture_probe probe.c)
target_include_directories(texture_probe PRIVATE "${VSL_WORKSPACE}/baselines/v0.15.0/src")
target_link_libraries(texture_probe PRIVATE "${VSL_WORKSPACE}/build-v015/Release/vsl_core.lib")
if(MSVC)
  target_compile_options(texture_probe PRIVATE /W4 /utf-8)
endif()
