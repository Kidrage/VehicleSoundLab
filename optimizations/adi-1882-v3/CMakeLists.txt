cmake_minimum_required(VERSION 3.21)
project(VehicleSound1882 VERSION 3.0 LANGUAGES C)
option(VSL_PROFILE "Record nested module timing" OFF)
option(VSL_STATIC_RUNTIME "Single static DSP runtime" ON)
option(VSL_LEGACY_MECHANICAL "Restore original independent mechanical chain" OFF)
option(VSL_FIXED_V10 "Compact production V10 with compiled immutable assets" OFF)
if(VSL_FIXED_V10 AND (NOT VSL_STATIC_RUNTIME OR VSL_LEGACY_MECHANICAL))
 message(FATAL_ERROR "Fixed V10 requires static runtime and baked mechanical colour")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
string(REPLACE "/DNDEBUG" "" CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE}")
include(sources.cmake)
add_library(vsl_core STATIC ${VSL_SOURCES})
if(VSL_FIXED_V10)
 target_sources(vsl_core PRIVATE embedded/fixed_assets.c)
 target_compile_definitions(vsl_core PUBLIC VSL_FIXED_V10)
 if(MSVC)
  target_compile_options(vsl_core PRIVATE /Gy /Gw)
 endif()
endif()
target_include_directories(vsl_core PUBLIC . modules src generated modules/sound_dna voices/events_v1)
target_compile_definitions(vsl_core PUBLIC VSL_DSP_APPROX VSL_DSP_FIR_TAPS=512 VSL_DSP_FAST_TRIG _CRT_SECURE_NO_WARNINGS=)
if(MSVC)
 target_compile_options(vsl_core PRIVATE /W4 /WX /utf-8 /fp:precise /FI"${CMAKE_CURRENT_SOURCE_DIR}/modules/dsp_support/fast_math.h")
endif()
add_library(vsl_source_bridge SHARED experiments/vehicle_source_v1/bridge.c)
target_link_libraries(vsl_source_bridge PRIVATE vsl_core)
set_target_properties(vsl_source_bridge PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
 target_link_options(vsl_source_bridge PRIVATE /MAP /OPT:REF /OPT:ICF)
endif()
if(MSVC)
 target_compile_options(vsl_source_bridge PRIVATE /W4 /WX /utf-8)
endif()
add_executable(render_cases tools/render_cases.c)
target_include_directories(render_cases PRIVATE . modules src generated modules/sound_dna voices/events_v1)
target_compile_options(render_cases PRIVATE /W4 /WX /utf-8)
enable_testing()
if(NOT VSL_FIXED_V10)
 add_executable(compile_fixed_assets tools/compile_fixed_assets.c)
 target_link_libraries(compile_fixed_assets PRIVATE vsl_core)
 target_include_directories(compile_fixed_assets PRIVATE experiments)
 target_compile_options(compile_fixed_assets PRIVATE /W4 /WX /utf-8)
foreach(unit checks input_checks event_checks policy_checks dna_checks state_dna_checks character_checks dynamics_checks)
 add_executable(${unit} experiments/vehicle_source_v1/${unit}.c)
 target_link_libraries(${unit} PRIVATE vsl_core)
 target_compile_options(${unit} PRIVATE /utf-8)
 if(NOT VSL_STATIC_RUNTIME)
  add_test(NAME ${unit} COMMAND ${unit})
  set_tests_properties(${unit} PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
 endif()
endforeach()
else()
 add_executable(fixed_memory_checks tests/fixed_memory_checks.c)
 target_link_libraries(fixed_memory_checks PRIVATE vsl_source_bridge vsl_core)
 target_compile_options(fixed_memory_checks PRIVATE /W4 /WX /utf-8)
 add_test(NAME fixed_v10_memory_contract COMMAND fixed_memory_checks)
endif()


option(VSL_CAPTURE_MODULES "Capture underlying module outputs" OFF)
target_sources(vsl_core PRIVATE dsp/diagnostics.c)
if(VSL_CAPTURE_MODULES)
 target_compile_definitions(vsl_core PUBLIC VSL_CAPTURE_MODULES)

endif()


target_sources(vsl_core PRIVATE dsp/perf.c)

if(VSL_PROFILE)
 target_compile_definitions(vsl_core PUBLIC VSL_PROFILE)
endif()
if(VSL_STATIC_RUNTIME)
 target_compile_definitions(vsl_core PUBLIC VSL_DSP_STATIC)
endif()
if(VSL_LEGACY_MECHANICAL)
 target_compile_definitions(vsl_core PUBLIC VSL_LEGACY_MECHANICAL)
endif()

add_executable(algorithm_checks tests/algorithm_checks.c)
target_link_libraries(algorithm_checks PRIVATE vsl_core)
target_compile_options(algorithm_checks PRIVATE /W4 /WX /utf-8)
add_test(NAME compact_dsp_algorithms COMMAND algorithm_checks)
