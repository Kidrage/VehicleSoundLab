cmake_minimum_required(VERSION 3.21)
project(VslDetailOscPreview LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(detail_osc STATIC stream.c ../osc_density_v1/model.c)
target_include_directories(detail_osc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../osc_density_v1")
add_executable(detail_osc_probe probe.c)
target_include_directories(detail_osc_probe PRIVATE "${VSL_ROOT}/baselines/v0.20.0/src")
target_link_libraries(detail_osc_probe PRIVATE detail_osc "${VSL_ROOT}/baselines/v0.20.0/bin/vsl_core.lib")
add_executable(detail_osc_checks checks.c)
target_link_libraries(detail_osc_checks PRIVATE detail_osc)
add_executable(detail_osc_preview WIN32 ui.c audio.c)
target_link_libraries(detail_osc_preview PRIVATE detail_osc winmm avrt comctl32 shell32)
enable_testing()
add_test(NAME detail_osc_stream COMMAND detail_osc_checks "${VSL_ROOT}/references/analysis/osc_quality_v2/rank64.model")
if(MSVC)
  foreach(target detail_osc detail_osc_probe detail_osc_checks detail_osc_preview)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
else()
  target_link_libraries(detail_osc PUBLIC m)
endif()
