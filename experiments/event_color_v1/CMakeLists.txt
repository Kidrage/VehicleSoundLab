cmake_minimum_required(VERSION 3.21)
project(VslEventColor LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(FROZEN "${VSL_ROOT}/baselines/v0.18.0/src")
function(hook marker replacement)
  string(LENGTH "${SOURCE}" before)
  string(REPLACE "${marker}" "" remaining "${SOURCE}")
  string(LENGTH "${remaining}" after)
  string(LENGTH "${marker}" length)
  math(EXPR removed "${before}-${after}")
  if(NOT removed EQUAL length)
    message(FATAL_ERROR "Missing or ambiguous frozen hook: ${marker}")
  endif()
  string(REPLACE "${marker}" "${replacement}" replaced "${SOURCE}")
  set(SOURCE "${replaced}" PARENT_SCOPE)
endfunction()
file(READ "${FROZEN}/core/engine.c" SOURCE)
string(FIND "${SOURCE}" "int vsl_process_layout(" A)
string(FIND "${SOURCE}" "void vsl_process(" B)
math(EXPR LENGTH "${B}-${A}")
string(SUBSTRING "${SOURCE}" ${A} ${LENGTH} SOURCE)
hook("int vsl_process_layout(VslEngine *e," "int vsl_color_process(VslEngine *e,VslEventColor *color,")
hook("double l,r,bed,event,spatial[VSL_MAX_CHANNELS],ir;" "double l,r,bed,event,spatial[VSL_MAX_CHANNELS],ir,main_pressure[2];")
hook("vsl_high_orders_tick(e,o);vsl_excitation_tick(e,o);"
  "vsl_high_orders_tick(e,o);main_pressure[0]=o->sample[0];main_pressure[1]=o->sample[1];vsl_excitation_tick(e,o);")
hook("vsl_character_tick(e,o);" "vsl_character_tick(e,o);vsl_color_tick(color,e,o,main_pressure);")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/color_engine.c"
  "#include \"color.h\"\n#include \"core/modules.h\"\n#include \"core/character.h\"\n#include \"drive/drive.h\"\n#include \"spatial/scene.h\"\n#include \"spatial/propagation.h\"\n#include <math.h>\n#include <string.h>\n${SOURCE}")
file(READ "${FROZEN}/session/take.c" SOURCE)
string(FIND "${SOURCE}" "unsigned vsl_take_process(" A)
string(SUBSTRING "${SOURCE}" ${A} -1 SOURCE)
hook("unsigned vsl_take_process(VslTakePlayer *player,VslEngine *e," "unsigned vsl_color_take(VslTakePlayer *player,VslEngine *e,VslEventColor *color,")
hook("vsl_process_layout(e,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);"
  "vsl_color_process(e,color,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/color_take.c" "#include \"color.h\"\n#include <math.h>\n${SOURCE}")
add_library(event_color STATIC split.c profile.c color.c "${CMAKE_CURRENT_BINARY_DIR}/color_engine.c" "${CMAKE_CURRENT_BINARY_DIR}/color_take.c")
target_include_directories(event_color PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${FROZEN}")
target_link_libraries(event_color PUBLIC "${VSL_ROOT}/baselines/v0.18.0/bin/vsl_core.lib")
add_executable(event_color_probe probe.c)
target_link_libraries(event_color_probe PRIVATE event_color)
add_executable(event_color_checks checks.c)
target_link_libraries(event_color_checks PRIVATE event_color)
enable_testing()
add_test(NAME main_pressure_event_color COMMAND event_color_checks)
if(MSVC)
  foreach(target event_color event_color_probe event_color_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
endif()
