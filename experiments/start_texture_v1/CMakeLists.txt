cmake_minimum_required(VERSION 3.21)
project(VslStartTexture1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}" drive)
add_subdirectory("${VSL_ROOT}/modules/sound_residual" sound-residual)
add_subdirectory("${VSL_ROOT}/modules/start_texture" start-texture)
add_executable(start_texture_checks checks.c)
target_link_libraries(start_texture_checks PRIVATE start_texture)
add_test(NAME startup_texture_gate_and_source COMMAND start_texture_checks)
add_executable(start_texture_probe probe.c)
target_link_libraries(start_texture_probe PRIVATE start_texture vsl_core)
foreach(target start_texture_checks start_texture_probe)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endif()
endforeach()
