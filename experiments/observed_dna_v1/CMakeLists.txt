cmake_minimum_required(VERSION 3.21)
project(VslObservedDna LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(observed_dna STATIC observation.c)
target_include_directories(observed_dna PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${VSL_ROOT}/baselines/v0.16.0/src")
target_link_libraries(observed_dna PUBLIC "${VSL_ROOT}/build-v016/Release/vsl_core.lib")
add_executable(observed_dna_probe probe.c)
target_link_libraries(observed_dna_probe PRIVATE observed_dna)
if(MSVC)
  target_compile_options(observed_dna PRIVATE /W4 /we4013 /utf-8)
  target_compile_options(observed_dna_probe PRIVATE /W4 /we4013 /utf-8)
endif()
