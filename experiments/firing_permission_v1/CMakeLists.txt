cmake_minimum_required(VERSION 3.21)
project(VslFiringPermission LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(VSL_PERMISSION_NATIVE "Link the current V017 core instead of frozen V016" OFF)
if(VSL_PERMISSION_NATIVE)
  set(VSL_INCLUDE "${VSL_ROOT}/src")
  set(VSL_LIBRARY "${VSL_ROOT}/build-v017/Release/vsl_core.lib")
else()
  set(VSL_INCLUDE "${VSL_ROOT}/baselines/v0.16.0/src")
  set(VSL_LIBRARY "${VSL_ROOT}/build-v016/Release/vsl_core.lib")
endif()
add_executable(firing_permission_probe probe.c)
target_include_directories(firing_permission_probe PRIVATE "${VSL_INCLUDE}")
target_link_libraries(firing_permission_probe PRIVATE "${VSL_LIBRARY}")
if(MSVC)
  target_compile_options(firing_permission_probe PRIVATE /W4 /we4013 /utf-8)
endif()
