cmake_minimum_required(VERSION 3.21)
project(VslStateContextExport LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(state_context_export export.c)
target_include_directories(state_context_export PRIVATE "${VSL_ROOT}/baselines/v0.16.0/src")
target_link_libraries(state_context_export PRIVATE "${VSL_ROOT}/build-v016/Release/vsl_core.lib")
if(MSVC)
  target_compile_options(state_context_export PRIVATE /W4 /we4013 /utf-8)
endif()
