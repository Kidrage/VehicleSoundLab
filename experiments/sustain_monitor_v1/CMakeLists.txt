cmake_minimum_required(VERSION 3.21)
project(VslSustainOscMonitor LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(FROZEN "${VSL_ROOT}/baselines/sustain_osc_v1/experiments/sustain_osc_v1")
add_library(sustain_monitor STATIC render.c "${FROZEN}/model.c" "${FROZEN}/source.c")
target_include_directories(sustain_monitor PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${FROZEN}")
add_executable(sustain_monitor_checks checks.c)
target_link_libraries(sustain_monitor_checks PRIVATE sustain_monitor)
add_executable(sustain_osc_monitor WIN32 ui.c audio.c smoke.c)
target_include_directories(sustain_osc_monitor PRIVATE "${VSL_ROOT}/baselines/v0.20.0/src")
target_link_libraries(sustain_osc_monitor PRIVATE sustain_monitor comctl32 shell32 winmm avrt "${VSL_ROOT}/baselines/v0.20.0/bin/vsl_core.lib")
enable_testing()
add_test(NAME monitor_stream_contract COMMAND sustain_monitor_checks "${VSL_ROOT}/references/analysis/sustain_osc_v1/voice.model")
foreach(target sustain_monitor sustain_monitor_checks sustain_osc_monitor)
  target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
endforeach()
