cmake_minimum_required(VERSION 3.21)
project(VslSceneMonitorV1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}" legacy EXCLUDE_FROM_ALL)
add_subdirectory("${VSL_ROOT}/modules/sound_dna" sound-dna)
add_subdirectory("${VSL_ROOT}/modules/dna_scene" dna-scene)
add_subdirectory("${VSL_ROOT}/modules/scene_monitor" scene-monitor)
add_executable(scene_monitor_checks checks.c recipe.c)
target_link_libraries(scene_monitor_checks PRIVATE scene_monitor)
add_test(NAME scene_monitor_contract COMMAND scene_monitor_checks
  "${VSL_ROOT}/voices/dna_bank_v1/V10 Exterior.model"
  "${VSL_ROOT}/voices/coherent_low_v1/measured_phase.model")
if(WIN32)
  add_executable(scene_monitor_ui WIN32 ui.c audio.c models.c recipe.c ui_checks.c)
  target_compile_definitions(scene_monitor_ui PRIVATE UNICODE _UNICODE)
  target_link_libraries(scene_monitor_ui PRIVATE scene_monitor comctl32 comdlg32 shell32 winmm avrt)
  add_executable(scene_monitor_device_verify verify.c audio.c recipe.c)
  target_link_libraries(scene_monitor_device_verify PRIVATE scene_monitor winmm avrt)
  foreach(target scene_monitor_ui scene_monitor_device_verify scene_monitor_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
endif()
