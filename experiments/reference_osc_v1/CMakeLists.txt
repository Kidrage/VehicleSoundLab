cmake_minimum_required(VERSION 3.21)
project(VslReferenceOsc LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(reference_osc STATIC model.c oscillator.c)
target_include_directories(reference_osc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(reference_osc_probe probe.c)
target_include_directories(reference_osc_probe PRIVATE "${VSL_ROOT}/baselines/v0.20.0/src")
target_link_libraries(reference_osc_probe PRIVATE reference_osc "${VSL_ROOT}/baselines/v0.20.0/bin/vsl_core.lib")
add_executable(reference_osc_checks checks.c)
target_link_libraries(reference_osc_checks PRIVATE reference_osc)
enable_testing()
add_test(NAME reference_osc_contracts COMMAND reference_osc_checks)
if(MSVC)
  foreach(target reference_osc reference_osc_probe reference_osc_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
else()
  target_link_libraries(reference_osc PUBLIC m)
endif()
