cmake_minimum_required(VERSION 3.21)
project(VslSoundDnaMonitor1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${VSL_ROOT}/modules/sound_dna" sound-dna-core)
add_library(sustain_monitor STATIC render.c)
target_include_directories(sustain_monitor PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(sustain_monitor PUBLIC sound_dna)
add_executable(sustain_monitor_checks checks.c)
add_executable(sound_dna_probe probe.c)
target_include_directories(sound_dna_probe PRIVATE "${VSL_ROOT}/baselines/v0.20.0/src")
target_link_libraries(sound_dna_probe PRIVATE sustain_monitor "${VSL_ROOT}/baselines/v0.20.0/bin/vsl_core.lib")
target_link_libraries(sustain_monitor_checks PRIVATE sustain_monitor)
add_executable(sustain_osc_monitor WIN32 ui.c audio.c smoke.c bank.c)
target_include_directories(sustain_osc_monitor PRIVATE "${VSL_ROOT}/baselines/v0.20.0/src")
target_link_libraries(sustain_osc_monitor PRIVATE sustain_monitor comctl32 shell32 winmm avrt comdlg32 "${VSL_ROOT}/baselines/v0.20.0/bin/vsl_core.lib")
enable_testing()
add_test(NAME monitor_stream_contract COMMAND sustain_monitor_checks "${VSL_ROOT}/references/analysis/sustain_osc_v1/voice.model")
foreach(target sustain_monitor sustain_monitor_checks sustain_osc_monitor sound_dna_probe)
  target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
endforeach()
