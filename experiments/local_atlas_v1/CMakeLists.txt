cmake_minimum_required(VERSION 3.21)
project(VslLocalAtlas LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/bandlimited_dna_v1" p)
add_library(local_atlas_assets STATIC assets.c schedule.c)
target_include_directories(local_atlas_assets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${VSL_ROOT}/src")
target_link_libraries(local_atlas_assets PUBLIC dna_atlas vsl_core)
add_executable(local_atlas_export probe.c)
target_link_libraries(local_atlas_export PRIVATE local_atlas_assets)
add_executable(local_atlas_checks checks.c)
target_link_libraries(local_atlas_checks PRIVATE local_atlas_assets)
add_executable(local_atlas_driver_checks driver_checks.c)
target_link_libraries(local_atlas_driver_checks PRIVATE local_atlas_assets atlas_drive)
foreach(target local_atlas_assets local_atlas_export local_atlas_checks local_atlas_driver_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  endif()
endforeach()
set(LA_CHECK_ROOT "${CMAKE_BINARY_DIR}/local-atlas-known" CACHE PATH "Generated known controls")
add_test(NAME local_atlas_transactional_inputs COMMAND local_atlas_checks
  "${LA_CHECK_ROOT}/known.manifest" "${LA_CHECK_ROOT}/known.trace"
  "${LA_CHECK_ROOT}/bad.manifest" "${LA_CHECK_ROOT}/bad.trace")
add_test(NAME local_atlas_same_driver_2_12 COMMAND local_atlas_driver_checks "${LA_CHECK_ROOT}/known.manifest")
set_tests_properties(local_atlas_transactional_inputs local_atlas_same_driver_2_12 PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
