cmake_minimum_required(VERSION 3.21)
project(VslStationaryBridge LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/handoff" previous-system)
add_executable(stationary_bridge_probe models.c probe.c)
target_link_libraries(stationary_bridge_probe PRIVATE atlas_take_io resonant_detail_io)
target_compile_features(stationary_bridge_probe PRIVATE c_std_11)
set_target_properties(stationary_bridge_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  target_compile_options(stationary_bridge_probe PRIVATE /W4 /we4013 /utf-8 /fp:strict)
endif()
