cmake_minimum_required(VERSION 3.21)
project(VslSustainOscExperiment LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(sustain_osc STATIC model.c source.c)
target_include_directories(sustain_osc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(sustain_osc_probe probe.c)
target_include_directories(sustain_osc_probe PRIVATE "${VSL_ROOT}/baselines/v0.20.0/src")
target_link_libraries(sustain_osc_probe PRIVATE sustain_osc "${VSL_ROOT}/baselines/v0.20.0/bin/vsl_core.lib")
add_executable(sustain_osc_checks checks.c)
target_link_libraries(sustain_osc_checks PRIVATE sustain_osc)
enable_testing()
add_test(NAME sustain_osc_contracts COMMAND sustain_osc_checks "${VSL_ROOT}/references/analysis/sustain_osc_v1/voice.model")
if(MSVC)
  foreach(target sustain_osc sustain_osc_probe sustain_osc_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
else()
  target_link_libraries(sustain_osc PUBLIC m)
endif()
