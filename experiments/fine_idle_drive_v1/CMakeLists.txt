cmake_minimum_required(VERSION 3.21)
project(VslFineIdleDrive1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/idle_drive_v1" previous-idle-drive)
add_executable(fine_idle_layer_checks layer_checks.c)
target_link_libraries(fine_idle_layer_checks PRIVATE fine_idle_models)
add_executable(fine_idle_owned_checks checks.c)
target_link_libraries(fine_idle_owned_checks PRIVATE atlas_take_io full_recipe_models fine_idle_models)
foreach(target fine_idle_layer_checks fine_idle_owned_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endforeach()
add_test(NAME fine_idle_layer_contract COMMAND fine_idle_layer_checks)
add_test(NAME fine_idle_owned_driving_take COMMAND fine_idle_owned_checks)
set_tests_properties(fine_idle_layer_contract fine_idle_owned_driving_take PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
if(WIN32)
  add_executable(fine_idle_drive_device device.c "${VSL_ROOT}/experiments/atlas_monitor_v1/audio.c")
  target_link_libraries(fine_idle_drive_device PRIVATE atlas_take_io full_recipe_models fine_idle_models win_output)
  target_compile_options(fine_idle_drive_device PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  set_target_properties(fine_idle_drive_device PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
