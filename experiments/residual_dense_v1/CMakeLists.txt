cmake_minimum_required(VERSION 3.21)
project(VslResidualDense1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${VSL_ROOT}/modules/sound_residual" sound-residual)
add_subdirectory("${VSL_ROOT}/modules/residual_dense" residual-dense)
add_library(dense_io STATIC "${VSL_ROOT}/src/io/file.c" "${VSL_ROOT}/src/io/wav.c")
target_include_directories(dense_io PUBLIC "${VSL_ROOT}/src")
add_executable(residual_dense_probe probe.c)
target_link_libraries(residual_dense_probe PRIVATE residual_dense dense_io)
add_executable(residual_dense_checks checks.c)
target_link_libraries(residual_dense_checks PRIVATE residual_dense)
foreach(target dense_io residual_dense_probe residual_dense_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endif()
endforeach()
enable_testing()
add_test(NAME residual_dense_contract COMMAND residual_dense_checks)
