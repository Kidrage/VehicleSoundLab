cmake_minimum_required(VERSION 3.21)
project(VslSpectralTexture1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/modal_texture_v1" modal-texture-study)
add_subdirectory("${VSL_ROOT}/modules/partitioned_fir" partitioned-fir)
add_subdirectory("${VSL_ROOT}/modules/spectral_texture" spectral-texture)
add_executable(spectral_texture_checks checks.c)
target_link_libraries(spectral_texture_checks PRIVATE spectral_texture_io)
add_executable(fine_idle_probe probe.c)
target_link_libraries(fine_idle_probe PRIVATE spectral_texture_io modal_texture_io modal_source_io modal_idle_io)
foreach(target spectral_texture_checks fine_idle_probe)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endforeach()
add_test(NAME spectral_FIR_source_contracts COMMAND spectral_texture_checks)
