cmake_minimum_required(VERSION 3.21)
project(VslSoundDnaExport LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${VSL_ROOT}/modules/sound_dna" sound-dna-core)
add_executable(sound_dna_export probe.c "${VSL_ROOT}/src/io/wav.c" "${VSL_ROOT}/src/io/file.c")
target_include_directories(sound_dna_export PRIVATE "${VSL_ROOT}/src")
target_link_libraries(sound_dna_export PRIVATE sound_dna)
target_compile_features(sound_dna_export PRIVATE c_std_11)
set_target_properties(sound_dna_export PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  target_compile_options(sound_dna_export PRIVATE /W4 /we4013 /utf-8 /fp:strict)
endif()
