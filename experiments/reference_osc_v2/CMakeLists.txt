cmake_minimum_required(VERSION 3.21)
project(VslTrackedOsc LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(V1 "${CMAKE_CURRENT_SOURCE_DIR}/../reference_osc_v1")
add_library(tracked_osc STATIC tracks.c pitch.c "${V1}/model.c" "${V1}/oscillator.c")
target_include_directories(tracked_osc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${V1}")
add_executable(tracked_osc_probe probe.c "${VSL_ROOT}/baselines/v0.20.0/src/io/wav.c" "${VSL_ROOT}/baselines/v0.20.0/src/io/file.c")
target_include_directories(tracked_osc_probe PRIVATE "${VSL_ROOT}/baselines/v0.20.0/src")
target_link_libraries(tracked_osc_probe PRIVATE tracked_osc)
add_executable(tracked_osc_checks checks.c)
target_link_libraries(tracked_osc_checks PRIVATE tracked_osc)
enable_testing()
add_test(NAME tracked_osc_contracts COMMAND tracked_osc_checks)
if(WIN32)
  add_executable(reference64_preview WIN32 ui.c audio.c)
  target_link_libraries(reference64_preview PRIVATE tracked_osc comctl32 shell32 winmm avrt)
  if(MSVC)
    target_compile_options(reference64_preview PRIVATE /W4 /we4013 /utf-8)
  endif()
endif()
if(MSVC)
  foreach(target tracked_osc tracked_osc_probe tracked_osc_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
else()
  target_link_libraries(tracked_osc PUBLIC m)
endif()
