cmake_minimum_required(VERSION 3.21)
project(VslEventSynth LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/modules/event_synth" event-module)
add_subdirectory("${VSL_ROOT}/modules/event_control" event-control-module)
add_executable(event_control_checks control_checks.c)
target_link_libraries(event_control_checks PRIVATE event_control)
add_executable(event_synth_checks checks.c)
target_link_libraries(event_synth_checks PRIVATE event_synth)
add_executable(event_contour_checks contour_checks.c)
target_link_libraries(event_contour_checks PRIVATE event_synth)
add_executable(event_synth_probe probe.c)
target_link_libraries(event_synth_probe PRIVATE event_synth)
target_include_directories(event_synth_probe PRIVATE "${VSL_ROOT}/voices/events_v1")
add_library(vsl_event_author_bridge SHARED author_bridge.c)
target_link_libraries(vsl_event_author_bridge PRIVATE event_synth)
set_target_properties(vsl_event_author_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  target_compile_options(vsl_event_author_bridge PRIVATE /W4 /we4013 /utf-8 /fp:strict)
endif()
foreach(target event_synth_checks event_synth_probe event_control_checks event_contour_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  endif()
endforeach()
add_test(NAME event_pool_lifecycle_and_identity COMMAND event_synth_checks)
add_test(NAME event_contour_clock_and_validation COMMAND event_contour_checks)
add_test(NAME event_semantic_control_and_scheduler COMMAND event_control_checks)
