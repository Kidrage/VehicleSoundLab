cmake_minimum_required(VERSION 3.21)
project(VslStartupCharacter LANGUAGES C)
enable_testing()
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${VSL_ROOT}/experiments/spectral_texture_v1" spectral-base)
add_library(startup_character STATIC "${VSL_ROOT}/modules/startup_character/voice.c" "${VSL_ROOT}/modules/startup_character/ridge.c")
target_sources(startup_character PRIVATE "${VSL_ROOT}/modules/crank_rotation/voice.c")
target_include_directories(startup_character PUBLIC "${VSL_ROOT}/modules")
target_link_libraries(startup_character PUBLIC spectral_texture)
add_library(startup_character_bridge SHARED bridge.c)
target_link_libraries(startup_character_bridge PRIVATE startup_character spectral_texture_io)
add_executable(crank_rotation_checks rotation_checks.c)
target_link_libraries(crank_rotation_checks PRIVATE startup_character)
add_test(NAME continuous_crank_rotation_contract COMMAND crank_rotation_checks)
foreach(t startup_character startup_character_bridge crank_rotation_checks)
 target_compile_features(${t} PRIVATE c_std_11)
 if(MSVC)
  target_compile_options(${t} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
 endif()
endforeach()
