cmake_minimum_required(VERSION 3.21)
project(VslJointCapture LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${VSL_ROOT}/handoff" base EXCLUDE_FROM_ALL)
add_executable(joint_capture probe.c session.c)
target_link_libraries(joint_capture PRIVATE catalog_drive_models atlas_take_io)
target_compile_features(joint_capture PRIVATE c_std_11)
set_target_properties(joint_capture PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  target_compile_options(joint_capture PRIVATE /W4 /we4013 /utf-8 /fp:strict)
endif()
