cmake_minimum_required(VERSION 3.21)
project(VslDriverStageCost1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${VSL_ROOT}/experiments/fine_idle_drive_v1" driver-base)
set(stage_functions sa_voice_tick_body sa_voice_tick_shaped_body drc_shape drc_voice_tick drs_voice_tick
  fi_state_tick ra_voice_tick da_voice_tick hb_mix_tick vsl_scene_tick vsl_drive_step vsl_orchestrate)
set(wrapped "/* Generated instrumentation only; original render source unchanged. */\n")
foreach(name IN LISTS stage_functions)
  string(APPEND wrapped "#define ${name} sp_${name}\n")
endforeach()
string(APPEND wrapped "#include \"${VSL_ROOT}/modules/atlas_drive/render.c\"\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/wrapped_render.c" "${wrapped}")
add_executable(stage_cost_plain probe.c wrappers.c)
add_executable(stage_cost_timed probe.c wrappers.c "${CMAKE_CURRENT_BINARY_DIR}/wrapped_render.c")
foreach(target stage_cost_plain stage_cost_timed)
  target_sources(${target} PRIVATE "${VSL_ROOT}/experiments/atlas_monitor_v1/audio.c")
  target_compile_features(${target} PRIVATE c_std_11)
  target_link_libraries(${target} PRIVATE atlas_take_io full_recipe_models fine_idle_models win_output)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  endif()
endforeach()
