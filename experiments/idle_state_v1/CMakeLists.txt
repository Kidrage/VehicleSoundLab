cmake_minimum_required(VERSION 3.21)
project(VslIdleState1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/handoff" handoff)
add_library(idle_state_source STATIC source.c "${VSL_ROOT}/experiments/modal_idle_v1/models.c")
target_link_libraries(idle_state_source PUBLIC modal_layer modal_source_io resonant_detail_io atlas_drive)
add_executable(idle_layer_checks kernel_checks.c)
target_link_libraries(idle_layer_checks PRIVATE modal_layer)
add_executable(idle_state_checks checks.c)
target_link_libraries(idle_state_checks PRIVATE idle_state_source full_recipe_models atlas_take)
add_executable(idle_state_probe probe.c)
target_link_libraries(idle_state_probe PRIVATE idle_state_source full_recipe_models)
set_target_properties(idle_layer_checks idle_state_checks idle_state_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(target idle_state_source idle_layer_checks idle_state_checks idle_state_probe)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endforeach()
add_test(NAME independent_modal_layer_contract COMMAND idle_layer_checks)
add_test(NAME idle_state_source_integration COMMAND idle_state_checks)
set_tests_properties(independent_modal_layer_contract idle_state_source_integration PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
