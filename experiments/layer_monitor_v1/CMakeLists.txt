cmake_minimum_required(VERSION 3.21)
project(VslLayerMonitor LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
set(LM_OLD_ROOT "" CACHE PATH "Protected old source, for baseline oracle only")
if(LM_OLD_ROOT)
  add_subdirectory("${LM_OLD_ROOT}/handoff" previous-system)
else()
  add_subdirectory("${VSL_ROOT}/experiments/fine_idle_drive_v1" previous-system)
endif()
add_executable(layer_canonical_probe probe.c)
target_link_libraries(layer_canonical_probe PRIVATE atlas_take_io full_recipe_models fine_idle_models)
target_compile_features(layer_canonical_probe PRIVATE c_std_11)
set_target_properties(layer_canonical_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(LM_OLD_ROOT)
  target_compile_definitions(layer_canonical_probe PRIVATE LM_LEGACY=1)
else()
  add_executable(layer_listen_checks checks.c)
  target_link_libraries(layer_listen_checks PRIVATE layer_monitor)
  target_compile_features(layer_listen_checks PRIVATE c_std_11)
  set_target_properties(layer_listen_checks PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  add_test(NAME layer_monitor_math_and_crossfade COMMAND layer_listen_checks)
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/probe-check")
  add_test(NAME layer_monitor_canonical_and_owned_replay COMMAND layer_canonical_probe "${CMAKE_BINARY_DIR}/probe-check")
  set_tests_properties(layer_monitor_canonical_and_owned_replay PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
  if(WIN32)
    add_executable(layer_monitor_device device.c "${VSL_ROOT}/experiments/atlas_monitor_v1/audio.c")
    target_link_libraries(layer_monitor_device PRIVATE atlas_take_io full_recipe_models fine_idle_models win_output)
    target_compile_options(layer_monitor_device PRIVATE /W4 /we4013 /utf-8 /fp:strict)
    set_target_properties(layer_monitor_device PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  endif()
endif()
if(MSVC)
  target_compile_options(layer_canonical_probe PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  if(NOT LM_OLD_ROOT)
    target_compile_options(layer_listen_checks PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  endif()
endif()
