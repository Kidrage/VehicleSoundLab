cmake_minimum_required(VERSION 3.21)
project(VslExhaustEnvelopeExperiment LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VSL_FROZEN "${VSL_ROOT}/baselines/v0.16.0/src")
# Copy the frozen scheduler with explicit per-instance transfer and tap args.
file(READ "${VSL_FROZEN}/core/engine.c" SOURCE)
string(FIND "${SOURCE}" "int vsl_process_layout(" START)
string(FIND "${SOURCE}" "void vsl_process(" END)
if(START EQUAL -1 OR END EQUAL -1 OR END LESS START)
  message(FATAL_ERROR "Frozen scheduler boundaries changed")
endif()
math(EXPR LENGTH "${END}-${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} SOURCE)
string(REPLACE "int vsl_process_layout(VslEngine *e," "int vsl_envelope_process_layout(VslEngine *e,VslEnvelopeTransfer *transfer," SOURCE "${SOURCE}")
string(REPLACE "int channels,float *stems)" "int channels,float *stems,float *taps)" SOURCE "${SOURCE}")
string(FIND "${SOURCE}" "vsl_excitation_tick(e,o);" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen excitation hook changed")
endif()
string(REPLACE "vsl_excitation_tick(e,o);" "vsl_envelope_excitation_tick(e,o,transfer,taps?taps+i*4:NULL);" SOURCE "${SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/envelope_engine.c" "#include \"transfer.h\"\n#include \"core/modules.h\"\n#include \"core/character.h\"\n#include \"drive/drive.h\"\n#include \"spatial/scene.h\"\n#include \"spatial/propagation.h\"\n#include <math.h>\n#include <string.h>\n${SOURCE}")
# Preserve fuel/lifecycle calculations, insert only after fired gating and
# before the independent pumping sum. Native source is never rewritten.
file(READ "${VSL_FROZEN}/core/excitation.c" SOURCE)
string(REPLACE "#include \"engine.h\"" "#include \"transfer.h\"" SOURCE "${SOURCE}")
string(REPLACE "void vsl_excitation_init(" "void vsl_envelope_cloned_excitation_init(" SOURCE "${SOURCE}")
string(REPLACE "void vsl_excitation_tick(VslEngine *e,VslObjectFrame *o)" "void vsl_envelope_excitation_tick(VslEngine *e,VslObjectFrame *o,VslEnvelopeTransfer *transfer,float *taps)" SOURCE "${SOURCE}")
string(REPLACE "    int side,n;" "    int side,n;\n    vsl_envelope_prepare(transfer,e);" SOURCE "${SOURCE}")
set(MARKER "        o->sample[VSL_EXHAUST_L+side]*=x->firing;")
string(FIND "${SOURCE}" "${MARKER}" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen fired/pumping boundary changed")
endif()
string(REPLACE "${MARKER}" "${MARKER}\n        if(taps)taps[side]=(float)o->sample[VSL_EXHAUST_L+side];\n        o->sample[VSL_EXHAUST_L+side]=vsl_envelope_side_tick(transfer,side,o->sample[VSL_EXHAUST_L+side]);\n        if(taps)taps[side+2]=(float)o->sample[VSL_EXHAUST_L+side];" SOURCE "${SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/envelope_excitation.c" "${SOURCE}")
file(READ "${VSL_FROZEN}/session/take.c" SOURCE)
string(FIND "${SOURCE}" "unsigned vsl_take_process(" START)
if(START EQUAL -1)
  message(FATAL_ERROR "Frozen take boundary changed")
endif()
string(SUBSTRING "${SOURCE}" ${START} -1 SOURCE)
string(REPLACE "unsigned vsl_take_process(VslTakePlayer *player,VslEngine *e," "unsigned vsl_envelope_take_process(VslTakePlayer *player,VslEngine *e,VslEnvelopeTransfer *transfer," SOURCE "${SOURCE}")
string(REPLACE "int channels,float *stems)" "int channels,float *stems,float *taps)" SOURCE "${SOURCE}")
set(MARKER "vsl_process_layout(e,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);")
string(FIND "${SOURCE}" "${MARKER}" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen take process call changed")
endif()
string(REPLACE "${MARKER}" "vsl_envelope_process_layout(e,transfer,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL,taps?taps+done*4:NULL);" SOURCE "${SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/envelope_take.c" "#include \"transfer.h\"\n#include <math.h>\n${SOURCE}")
add_library(exhaust_envelope STATIC biquad.c profile.c transfer.c "${CMAKE_CURRENT_BINARY_DIR}/envelope_engine.c" "${CMAKE_CURRENT_BINARY_DIR}/envelope_excitation.c" "${CMAKE_CURRENT_BINARY_DIR}/envelope_take.c")
target_include_directories(exhaust_envelope PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${VSL_FROZEN}" "${VSL_FROZEN}/core")
target_link_libraries(exhaust_envelope PUBLIC "${VSL_ROOT}/build-v016/Release/vsl_core.lib")
add_executable(exhaust_envelope_probe probe.c)
target_link_libraries(exhaust_envelope_probe PRIVATE exhaust_envelope)
add_executable(exhaust_envelope_checks checks.c)
target_link_libraries(exhaust_envelope_checks PRIVATE exhaust_envelope)
enable_testing()
add_test(NAME fired_exhaust_envelope_contracts COMMAND exhaust_envelope_checks)
if(MSVC)
  foreach(target exhaust_envelope exhaust_envelope_probe exhaust_envelope_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
endif()
