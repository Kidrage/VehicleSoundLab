cmake_minimum_required(VERSION 3.21)
project(VslIdleDrive1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/idle_state_v1" previous-idle)
add_executable(idle_control_checks control_checks.c)
target_link_libraries(idle_control_checks PRIVATE modal_layer)
add_executable(idle_owned_checks checks.c "${VSL_ROOT}/experiments/idle_state_v1/source.c")
target_link_libraries(idle_owned_checks PRIVATE atlas_take_io full_recipe_models idle_drive_models)
foreach(target idle_control_checks idle_owned_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endforeach()
add_test(NAME fixed_idle_amount_control COMMAND idle_control_checks)
add_test(NAME fixed_idle_owned_driving_take COMMAND idle_owned_checks)
set_tests_properties(fixed_idle_amount_control fixed_idle_owned_driving_take PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
if(WIN32)
  add_executable(idle_drive_device device.c "${VSL_ROOT}/experiments/atlas_monitor_v1/audio.c")
  target_link_libraries(idle_drive_device PRIVATE atlas_take_io full_recipe_models idle_drive_models win_output)
  target_compile_options(idle_drive_device PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  set_target_properties(idle_drive_device PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
