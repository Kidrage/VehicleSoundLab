cmake_minimum_required(VERSION 3.21)
project(VslPressureSampling LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(pressure_sampling_probe probe.c)
target_include_directories(pressure_sampling_probe PRIVATE "${VSL_ROOT}/baselines/v0.17.0/src")
target_link_libraries(pressure_sampling_probe PRIVATE "${VSL_ROOT}/baselines/v0.17.0/bin/vsl_core.lib")
if(MSVC)
  target_compile_options(pressure_sampling_probe PRIVATE /W4 /we4013 /utf-8)
endif()
