cmake_minimum_required(VERSION 3.21)
project(VslStatePassage1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${VSL_ROOT}/modules/sound_dna" sound-dna)
add_subdirectory("${VSL_ROOT}/modules/harmonic_body" harmonic-body)
add_subdirectory("${VSL_ROOT}/modules/dna_atlas" dna-atlas)
add_library(passage_io STATIC "${VSL_ROOT}/src/io/file.c" "${VSL_ROOT}/src/io/wav.c")
target_include_directories(passage_io PUBLIC "${VSL_ROOT}/src")
add_executable(state_passage_probe probe.c trajectory.c)
target_link_libraries(state_passage_probe PRIVATE dna_atlas passage_io)
foreach(target passage_io state_passage_probe)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endif()
endforeach()
