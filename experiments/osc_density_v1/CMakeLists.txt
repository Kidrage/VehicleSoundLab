cmake_minimum_required(VERSION 3.21)
project(VslDenseOscDiagnostic LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(dense_osc STATIC model.c oscillator.c)
target_include_directories(dense_osc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(dense_osc_probe probe.c)
target_include_directories(dense_osc_probe PRIVATE "${VSL_ROOT}/baselines/v0.20.0/src")
target_link_libraries(dense_osc_probe PRIVATE dense_osc "${VSL_ROOT}/baselines/v0.20.0/bin/vsl_core.lib")
if(MSVC)
  foreach(target dense_osc dense_osc_probe)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
else()
  target_link_libraries(dense_osc PUBLIC m)
endif()
