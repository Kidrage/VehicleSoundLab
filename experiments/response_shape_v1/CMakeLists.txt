cmake_minimum_required(VERSION 3.21)
project(VslResponseShape LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(FROZEN "${VSL_ROOT}/baselines/v0.17.0/src")
function(hook marker replacement)
  string(LENGTH "${SOURCE}" before)
  string(REPLACE "${marker}" "" remaining "${SOURCE}")
  string(LENGTH "${remaining}" after)
  string(LENGTH "${marker}" length)
  math(EXPR removed "${before}-${after}")
  if(NOT removed EQUAL length)
    message(FATAL_ERROR "Missing or ambiguous frozen hook: ${marker}")
  endif()
  string(REPLACE "${marker}" "${replacement}" replaced "${SOURCE}")
  set(SOURCE "${replaced}" PARENT_SCOPE)
endfunction()
file(READ "${FROZEN}/core/character.c" SOURCE)
string(FIND "${SOURCE}" "static double random_bipolar(" A)
string(FIND "${SOURCE}" "void vsl_character_init(" B)
math(EXPR LENGTH "${B}-${A}")
string(SUBSTRING "${SOURCE}" ${A} ${LENGTH} RANDOM)
string(FIND "${SOURCE}" "void vsl_character_tick(" A)
string(SUBSTRING "${SOURCE}" ${A} -1 SOURCE)
hook("void vsl_character_tick(VslEngine *e,VslObjectFrame *o)" "void vsl_response_character_tick(VslEngine *e,VslResponseShape *response,VslObjectFrame *o)")
hook("vsl_character_pressure_strike(&s->pressure,engine->bank[c],pressure_gain,shape,elapsed);"
  "vsl_response_pressure_strike(response,e,&s->pressure,engine->bank[c],pressure_gain,shape,elapsed);")
hook("vsl_character_structure_strike(&s->structure,structure_gain,shape,elapsed);"
  "vsl_response_structure_strike(response,e,&s->structure,structure_gain,shape,elapsed);")
hook("vsl_character_pressure_init(&s->pressure,e->sample_rate);"
  "vsl_response_clear(response);vsl_character_pressure_init(&s->pressure,e->sample_rate);")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/response_character.c"
  "#include \"response.h\"\n#include \"core/modules.h\"\n#include \"profiles/firing.h\"\n#include \"drive/fueling.h\"\n#include <math.h>\n${RANDOM}${SOURCE}")
file(READ "${FROZEN}/core/engine.c" SOURCE)
string(FIND "${SOURCE}" "int vsl_process_layout(" A)
string(FIND "${SOURCE}" "void vsl_process(" B)
math(EXPR LENGTH "${B}-${A}")
string(SUBSTRING "${SOURCE}" ${A} ${LENGTH} SOURCE)
hook("int vsl_process_layout(VslEngine *e," "int vsl_response_process(VslEngine *e,VslResponseShape *response,")
hook("vsl_character_tick(e,o);" "vsl_response_begin(response);vsl_response_character_tick(e,response,o);vsl_response_apply(response,e,o);")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/response_engine.c"
  "#include \"response.h\"\n#include \"core/modules.h\"\n#include \"drive/drive.h\"\n#include \"spatial/scene.h\"\n#include <math.h>\n#include <string.h>\n${SOURCE}")
file(READ "${FROZEN}/session/take.c" SOURCE)
string(FIND "${SOURCE}" "unsigned vsl_take_process(" A)
string(SUBSTRING "${SOURCE}" ${A} -1 SOURCE)
hook("unsigned vsl_take_process(VslTakePlayer *player,VslEngine *e," "unsigned vsl_response_take(VslTakePlayer *player,VslEngine *e,VslResponseShape *response,")
hook("vsl_process_layout(e,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);"
  "vsl_response_process(e,response,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/response_take.c" "#include \"response.h\"\n#include <math.h>\n${SOURCE}")
add_library(response_shape STATIC kernel.c profile.c response.c
  "${CMAKE_CURRENT_BINARY_DIR}/response_character.c"
  "${CMAKE_CURRENT_BINARY_DIR}/response_engine.c"
  "${CMAKE_CURRENT_BINARY_DIR}/response_take.c")
