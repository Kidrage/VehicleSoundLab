cmake_minimum_required(VERSION 3.21)
project(VslDriveCatalog LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/layer_monitor_v1" previous-system)
add_executable(catalog_contract_checks contract.c)
target_link_libraries(catalog_contract_checks PRIVATE catalog_drive_models)
add_executable(catalog_render_checks render_checks.c)
target_link_libraries(catalog_render_checks PRIVATE catalog_drive_models atlas_take_io)
add_executable(catalog_recipe_checks recipe_checks.c)
target_link_libraries(catalog_recipe_checks PRIVATE catalog_drive_models)
foreach(target catalog_contract_checks catalog_render_checks catalog_recipe_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  endif()
endforeach()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/catalog-check")
add_test(NAME catalog_owned_configs_and_capabilities COMMAND catalog_contract_checks)
add_test(NAME catalog_all_voice_driving_and_replay COMMAND catalog_render_checks "${CMAKE_BINARY_DIR}/catalog-check")
add_test(NAME catalog_stereo_recipe_coupling COMMAND catalog_recipe_checks)
set_tests_properties(catalog_owned_configs_and_capabilities catalog_all_voice_driving_and_replay catalog_stereo_recipe_coupling PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
