cmake_minimum_required(VERSION 3.21)
project(VslFiringPulse LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(FROZEN "${VSL_ROOT}/baselines/v0.19.0/src")
function(hook marker replacement)
  string(LENGTH "${SOURCE}" before)
  string(REPLACE "${marker}" "" remaining "${SOURCE}")
  string(LENGTH "${remaining}" after)
  string(LENGTH "${marker}" length)
  math(EXPR removed "${before}-${after}")
  if(NOT removed EQUAL length)
    message(FATAL_ERROR "Missing/ambiguous frozen hook: ${marker}")
  endif()
  string(REPLACE "${marker}" "${replacement}" replaced "${SOURCE}")
  set(SOURCE "${replaced}" PARENT_SCOPE)
endfunction()
file(READ "${FROZEN}/core/engine.c" SOURCE)
string(FIND "${SOURCE}" "int vsl_process_layout(" A)
string(FIND "${SOURCE}" "void vsl_process(" B)
math(EXPR LENGTH "${B}-${A}")
string(SUBSTRING "${SOURCE}" ${A} ${LENGTH} SOURCE)
hook("int vsl_process_layout(VslEngine *e," "int fp_process(VslEngine *e,FpAdapter *s,")
hook("ir=vsl_ir_tick(e," "fp_tick(s,e,o,main_pressure);ir=vsl_ir_tick(e,")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pulse_engine.c"
  "#include \"adapter.h\"\n#include \"core/modules.h\"\n#include \"core/character.h\"\n#include \"core/event_color.h\"\n#include \"drive/drive.h\"\n#include \"spatial/scene.h\"\n#include \"spatial/propagation.h\"\n#include <math.h>\n#include <string.h>\n${SOURCE}")
file(READ "${FROZEN}/session/take.c" SOURCE)
string(FIND "${SOURCE}" "unsigned vsl_take_process(" A)
string(SUBSTRING "${SOURCE}" ${A} -1 SOURCE)
hook("unsigned vsl_take_process(VslTakePlayer *player,VslEngine *e," "unsigned fp_take(VslTakePlayer *player,VslEngine *e,FpAdapter *s,")
hook("vsl_process_layout(e,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);"
  "fp_process(e,s,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pulse_take.c" "#include \"adapter.h\"\n#include <math.h>\n${SOURCE}")
add_library(firing_pulse STATIC profile.c kernel.c adapter.c "${CMAKE_CURRENT_BINARY_DIR}/pulse_engine.c" "${CMAKE_CURRENT_BINARY_DIR}/pulse_take.c")
target_include_directories(firing_pulse PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${FROZEN}")
target_link_libraries(firing_pulse PUBLIC "${VSL_ROOT}/baselines/v0.19.0/bin/vsl_core.lib")
add_executable(firing_pulse_probe probe.c)
target_link_libraries(firing_pulse_probe PRIVATE firing_pulse)
add_executable(firing_pulse_checks checks.c)
target_link_libraries(firing_pulse_checks PRIVATE firing_pulse)
enable_testing()
add_test(NAME firing_pulse_contracts COMMAND firing_pulse_checks)
if(MSVC)
  foreach(target firing_pulse firing_pulse_probe firing_pulse_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
endif()
