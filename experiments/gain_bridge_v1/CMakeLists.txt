cmake_minimum_required(VERSION 3.21)
project(VslGainBridge LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VSL_FROZEN "${VSL_ROOT}/baselines/v0.16.0/src")
file(READ "${VSL_FROZEN}/core/engine.c" SOURCE)
string(FIND "${SOURCE}" "int vsl_process_layout(" START)
string(FIND "${SOURCE}" "void vsl_process(" END)
if(START EQUAL -1 OR END EQUAL -1 OR END LESS START)
  message(FATAL_ERROR "Frozen scheduler boundaries missing")
endif()
math(EXPR LENGTH "${END}-${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} SOURCE)
string(REPLACE "int vsl_process_layout(VslEngine *e," "int vsl_gain_bridge_process(VslEngine *e,VslGainBridge *bridge," SOURCE "${SOURCE}")
string(REPLACE "int channels,float *stems)" "int channels,float *stems,float *gain_pairs)" SOURCE "${SOURCE}")
set(MARKER "vsl_orchestrate(e,e->control_period/e->sample_rate);vsl_events_update(e);")
string(FIND "${SOURCE}" "${MARKER}" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen voice-control hook missing")
endif()
string(REPLACE "${MARKER}" "${MARKER}vsl_gain_bridge_control(bridge,&e->voice,e->control_period);" SOURCE "${SOURCE}")
set(MARKER "        vsl_clocks_tick(e);")
string(FIND "${SOURCE}" "${MARKER}" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen audio-voice hook missing")
endif()
string(REPLACE "${MARKER}" "        VslVoicing control_voice=e->voice;\n        vsl_gain_bridge_audio(bridge,&e->voice,gain_pairs?gain_pairs+i*VSL_BRIDGE_GAINS*2:NULL);\n${MARKER}" SOURCE "${SOURCE}")
string(REPLACE "vsl_scene_tick(e,o,spatial,channels);++e->frame;" "vsl_scene_tick(e,o,spatial,channels);e->voice=control_voice;++e->frame;" SOURCE "${SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bridge_engine.c" "#include \"bridge.h\"\n#include \"core/modules.h\"\n#include \"core/character.h\"\n#include \"drive/drive.h\"\n#include \"spatial/scene.h\"\n#include <math.h>\n#include <string.h>\n${SOURCE}")
file(READ "${VSL_FROZEN}/session/take.c" SOURCE)
string(FIND "${SOURCE}" "unsigned vsl_take_process(" START)
if(START EQUAL -1)
  message(FATAL_ERROR "Frozen take process missing")
endif()
string(SUBSTRING "${SOURCE}" ${START} -1 SOURCE)
string(REPLACE "unsigned vsl_take_process(VslTakePlayer *player,VslEngine *e," "unsigned vsl_gain_bridge_take(VslTakePlayer *player,VslEngine *e,VslGainBridge *bridge," SOURCE "${SOURCE}")
string(REPLACE "int channels,float *stems)" "int channels,float *stems,float *gain_pairs)" SOURCE "${SOURCE}")
set(MARKER "vsl_process_layout(e,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);")
string(FIND "${SOURCE}" "${MARKER}" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen take dispatch hook missing")
endif()
string(REPLACE "${MARKER}" "vsl_gain_bridge_process(e,bridge,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL,gain_pairs?gain_pairs+done*VSL_BRIDGE_GAINS*2:NULL);" SOURCE "${SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bridge_take.c" "#include \"bridge.h\"\n#include <math.h>\n${SOURCE}")
add_library(gain_bridge STATIC bridge.c "${CMAKE_CURRENT_BINARY_DIR}/bridge_engine.c" "${CMAKE_CURRENT_BINARY_DIR}/bridge_take.c")
target_include_directories(gain_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${VSL_FROZEN}")
target_link_libraries(gain_bridge PUBLIC "${VSL_ROOT}/build-v016/Release/vsl_core.lib")
add_executable(gain_bridge_probe probe.c)
target_link_libraries(gain_bridge_probe PRIVATE gain_bridge)
add_executable(gain_bridge_checks checks.c)
target_link_libraries(gain_bridge_checks PRIVATE gain_bridge)
enable_testing()
add_test(NAME gain_bridge_contracts COMMAND gain_bridge_checks)
if(MSVC)
  foreach(target gain_bridge gain_bridge_probe gain_bridge_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
endif()
