cmake_minimum_required(VERSION 3.21)
project(VslIdleMicrotexture1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/spectral_texture_v1" fine-static)
add_subdirectory("${VSL_ROOT}/modules/spectral_motion" spectral-motion)
add_executable(spectral_motion_checks checks.c)
target_link_libraries(spectral_motion_checks PRIVATE spectral_motion)
add_executable(fast_idle_probe probe.c)
target_link_libraries(fast_idle_probe PRIVATE spectral_motion spectral_texture_io modal_texture_io modal_source_io modal_idle_io)
foreach(target spectral_motion_checks fast_idle_probe)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endforeach()
add_test(NAME spectral_correlated_amplitude_contracts COMMAND spectral_motion_checks)
if(WIN32)
  add_subdirectory("${VSL_ROOT}/modules/win_output" win-output)
  add_library(fast_idle_monitor STATIC models.c render.c)
  target_link_libraries(fast_idle_monitor PUBLIC spectral_motion spectral_texture_io modal_texture_io modal_source_io)
  add_executable(fast_idle_device device_check.c)
  target_link_libraries(fast_idle_device PRIVATE fast_idle_monitor win_output modal_idle_io)
  add_executable(fast_idle_monitor_checks monitor_checks.c)
  target_link_libraries(fast_idle_monitor_checks PRIVATE fast_idle_monitor)
  add_test(NAME fixed_idle_monitor_control_ramps COMMAND fast_idle_monitor_checks)
  add_executable(VslFastIdleMonitor WIN32 ui.c)
  target_compile_definitions(VslFastIdleMonitor PRIVATE UNICODE _UNICODE)
  target_link_libraries(VslFastIdleMonitor PRIVATE fast_idle_monitor win_output comctl32 shell32)
  foreach(target fast_idle_monitor fast_idle_device fast_idle_monitor_checks VslFastIdleMonitor)
    target_compile_features(${target} PRIVATE c_std_11)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
    endif()
  endforeach()
endif()
