cmake_minimum_required(VERSION 3.21)
project(VslFirMac1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/modules/partitioned_fir" partitioned-fir)
add_subdirectory("${VSL_ROOT}/modules/resonant_detail" resonant-detail)
add_subdirectory("${VSL_ROOT}/modules/spectral_texture" spectral-texture)
add_library(partitioned_fir_trial STATIC filter_sse2.c mac.c)
target_include_directories(partitioned_fir_trial PUBLIC "${VSL_ROOT}/modules")
target_link_libraries(partitioned_fir_trial PUBLIC partitioned_fir)
add_library(partitioned_fir_scalar STATIC filter_sse2.c mac.c)
target_link_libraries(partitioned_fir_scalar PUBLIC partitioned_fir)
target_compile_definitions(partitioned_fir_scalar PRIVATE PF_FORCE_SCALAR=1)
target_include_directories(partitioned_fir_scalar PUBLIC "${VSL_ROOT}/modules")
if(VSL_FIR_REFERENCE_FILE)
  add_library(fir_reference OBJECT "${VSL_FIR_REFERENCE_FILE}")
  target_include_directories(fir_reference PRIVATE "${VSL_ROOT}/modules/partitioned_fir")
  target_compile_definitions(fir_reference PRIVATE
    pf_kernel_create=reference_pf_kernel_create pf_kernel_destroy=reference_pf_kernel_destroy
    pf_state_create=reference_pf_state_create pf_state_destroy=reference_pf_state_destroy
    pf_filter_block=reference_pf_filter_block)
endif()
foreach(kind optimized scalar)
  add_executable(fir_${kind}_checks probe.c)
  if(kind STREQUAL "scalar")
    target_link_libraries(fir_${kind}_checks PRIVATE partitioned_fir_scalar spectral_texture_io)
  else()
    target_link_libraries(fir_${kind}_checks PRIVATE partitioned_fir_trial spectral_texture_io)
  endif()
  if(VSL_FIR_REFERENCE_FILE)
    target_sources(fir_${kind}_checks PRIVATE $<TARGET_OBJECTS:fir_reference>)
    target_compile_definitions(fir_${kind}_checks PRIVATE PF_HAVE_REFERENCE=1)
  endif()
  add_test(NAME fir_${kind}_contracts COMMAND fir_${kind}_checks)
endforeach()
foreach(target partitioned_fir_trial partitioned_fir_scalar fir_optimized_checks fir_scalar_checks fir_reference)
  if(TARGET ${target})
    target_compile_features(${target} PRIVATE c_std_11)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    endif()
  endif()
endforeach()
