cmake_minimum_required(VERSION 3.21)
project(VslVehicleSource LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/catalog_drive_v1" existing-catalog)
add_subdirectory("${VSL_ROOT}/modules/vehicle_input" input-module)
add_subdirectory("${VSL_ROOT}/modules/event_synth" event-module)
add_subdirectory("${VSL_ROOT}/modules/event_control" event-control-module)
add_subdirectory("${VSL_ROOT}/modules/state_dna" state-dna-module)
add_subdirectory("${VSL_ROOT}/modules/dna_character" dna-character-module)
add_subdirectory("${VSL_ROOT}/modules/startup_character" startup-character-module)
add_subdirectory("${VSL_ROOT}/modules/startup_flare" startup-flare-module)
add_subdirectory("${VSL_ROOT}/modules/vehicle_source" source-module)
add_library(vsl_source_bridge SHARED bridge.c)
target_link_libraries(vsl_source_bridge PRIVATE vehicle_source catalog_drive_models)
add_executable(vehicle_source_checks checks.c)
target_link_libraries(vehicle_source_checks PRIVATE vehicle_source catalog_drive_models)
add_executable(vehicle_source_probe probe.c bridge.c)
target_link_libraries(vehicle_source_probe PRIVATE vehicle_source catalog_drive_models)
add_executable(vehicle_input_checks input_checks.c)
target_link_libraries(vehicle_input_checks PRIVATE vehicle_source catalog_drive_models)
add_executable(vehicle_event_checks event_checks.c)
target_link_libraries(vehicle_event_checks PRIVATE vehicle_source catalog_drive_models)
add_executable(event_policy_checks policy_checks.c)
add_executable(vehicle_dna_checks dna_checks.c)
add_executable(vehicle_state_dna_checks state_dna_checks.c)
add_executable(vehicle_character_checks character_checks.c)
target_link_libraries(vehicle_character_checks PRIVATE vehicle_source catalog_drive_models)
target_link_libraries(vehicle_state_dna_checks PRIVATE vehicle_source catalog_drive_models)
target_link_libraries(vehicle_dna_checks PRIVATE vehicle_source catalog_drive_models)
target_link_libraries(event_policy_checks PRIVATE event_control)
foreach(target vsl_source_bridge vehicle_source_checks vehicle_source_probe vehicle_input_checks vehicle_event_checks event_policy_checks vehicle_dna_checks vehicle_state_dna_checks vehicle_character_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  endif()
endforeach()
if(MSVC)
  target_compile_options(event_policy_checks PRIVATE /UNDEBUG)
else()
  target_compile_options(event_policy_checks PRIVATE -UNDEBUG)
endif()
add_test(NAME vehicle_event_policy_recipe COMMAND event_policy_checks)
add_test(NAME authored_dna_owned_assets COMMAND vehicle_dna_checks)
add_test(NAME state_dna_common_clock_and_owned_assets COMMAND vehicle_state_dna_checks)
set_tests_properties(authored_dna_owned_assets PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
add_test(NAME independent_source_stems_and_spatial_bypass COMMAND vehicle_source_checks)
add_test(NAME vehicle_external_input_authority COMMAND vehicle_input_checks)
add_test(NAME vehicle_event_native_clock_and_stems COMMAND vehicle_event_checks)
set_tests_properties(vehicle_event_native_clock_and_stems PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
set_tests_properties(vehicle_external_input_authority PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
set_tests_properties(independent_source_stems_and_spatial_bypass PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
add_test(NAME unified_dna_roles_and_lifecycle COMMAND vehicle_character_checks)
set_tests_properties(unified_dna_roles_and_lifecycle PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
