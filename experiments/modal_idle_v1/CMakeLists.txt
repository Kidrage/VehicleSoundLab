cmake_minimum_required(VERSION 3.21)
project(VslModalIdle1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${VSL_ROOT}/modules/modal_source" modal-source)
add_subdirectory("${VSL_ROOT}/modules/resonant_detail" resonant-detail)
add_subdirectory("${VSL_ROOT}/modules/resonant_detail_io" resonant-detail-io)
add_library(modal_idle_io STATIC "${VSL_ROOT}/src/io/file.c" "${VSL_ROOT}/src/io/wav.c")
target_include_directories(modal_idle_io PUBLIC "${VSL_ROOT}/src")
add_library(modal_idle_models STATIC models.c)
target_link_libraries(modal_idle_models PUBLIC modal_source_io resonant_detail_io modal_idle_io)
add_executable(modal_idle_probe probe.c)
target_link_libraries(modal_idle_probe PRIVATE modal_idle_models)
add_executable(modal_idle_checks checks.c)
target_link_libraries(modal_idle_checks PRIVATE modal_idle_models)
foreach(target modal_idle_io modal_idle_models modal_idle_probe modal_idle_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endforeach()
enable_testing()
add_test(NAME independent_modal_source COMMAND modal_idle_checks)
add_test(NAME fixed_idle_parameter_contract COMMAND modal_idle_checks
  "${CMAKE_CURRENT_SOURCE_DIR}/F2_fixed_idle.msource" "${CMAKE_CURRENT_SOURCE_DIR}/F2_fixed_idle.rdetail")
