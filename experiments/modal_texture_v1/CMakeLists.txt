cmake_minimum_required(VERSION 3.21)
project(VslModalTexture1 LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/modal_idle_v1" modal-idle)
add_subdirectory("${VSL_ROOT}/modules/modal_texture" modal-texture)
add_executable(modal_texture_checks checks.c)
target_link_libraries(modal_texture_checks PRIVATE modal_texture_io)
add_executable(modal_texture_probe probe.c)
target_link_libraries(modal_texture_probe PRIVATE modal_texture_io modal_source_io modal_idle_io)
foreach(target modal_texture_checks modal_texture_probe)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endforeach()
add_test(NAME modal_amplitude_controls COMMAND modal_texture_checks)
