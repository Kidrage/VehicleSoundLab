cmake_minimum_required(VERSION 3.21)
project(VslCatchBloom1 LANGUAGES C)
enable_testing()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../handoff" system)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../modules/catch_bloom" bloom)
add_executable(catch_bloom_probe probe.c)
target_link_libraries(catch_bloom_probe PRIVATE atlas_drive catch_bloom)
add_executable(catch_bloom_checks checks.c)
target_link_libraries(catch_bloom_checks PRIVATE catch_bloom)
add_test(NAME catch_bloom_contracts COMMAND catch_bloom_checks)
foreach(target catch_bloom_probe catch_bloom_checks)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endif()
endforeach()
