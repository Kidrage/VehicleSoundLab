cmake_minimum_required(VERSION 3.21)
project(VslSourceAudition LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VSL_FROZEN "${VSL_ROOT}/baselines/v0.16.0/src")
# Explicit observer state in copies; never rewrite native source or binaries.
file(READ "${VSL_FROZEN}/spatial/scene.c" SOURCE)
string(FIND "${SOURCE}" "void vsl_scene_update(" START)
if(START EQUAL -1)
  message(FATAL_ERROR "Frozen scene update boundary missing")
endif()
string(SUBSTRING "${SOURCE}" ${START} -1 SOURCE)
string(REPLACE "void vsl_scene_update(VslEngine *e,int channels)" "void vsl_audition_scene_update(VslEngine *e,VslSpatial *s,int channels)" SOURCE "${SOURCE}")
string(REPLACE "VslSpatial *s=&e->spatial; int i,c;" "int i,c;" SOURCE "${SOURCE}")
string(REPLACE "void vsl_scene_tick(VslEngine *e,const VslObjectFrame *o,double *output,int channels)" "void vsl_audition_scene_tick(VslEngine *e,VslSpatial *s,const VslObjectFrame *o,unsigned mask,double *output,int channels)" SOURCE "${SOURCE}")
string(REPLACE "VslSpatial *s=&e->spatial;int i,c;" "int i,c;" SOURCE "${SOURCE}")
string(REPLACE "vsl_scene_update(e,channels);" "vsl_audition_scene_update(e,s,channels);" SOURCE "${SOURCE}")
string(REPLACE "&objects[i]" "vsl_object_info(i)" SOURCE "${SOURCE}")
foreach(axis x y z)
  string(REPLACE "objects[i].${axis}" "vsl_object_info(i)->${axis}" SOURCE "${SOURCE}")
endforeach()
set(MARKER "        double x=o->sample[i],hi,paths[VSL_PATH_COUNT],blend=s->propagation.mix;")
string(FIND "${SOURCE}" "${MARKER}" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen scene source loop missing")
endif()
string(REPLACE "${MARKER}" "        if(!(mask&(1u<<i)))continue;\n${MARKER}" SOURCE "${SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/audition_scene.c" "#include \"audition.h\"\n#include \"spatial/scene.h\"\n#include \"spatial/propagation.h\"\n#include \"core/modules.h\"\n#include <math.h>\n#include <string.h>\n${SOURCE}")
file(READ "${VSL_FROZEN}/core/engine.c" SOURCE)
string(FIND "${SOURCE}" "int vsl_process_layout(" START)
string(FIND "${SOURCE}" "void vsl_process(" END)
if(START EQUAL -1 OR END EQUAL -1 OR END LESS START)
  message(FATAL_ERROR "Frozen scheduler boundaries missing")
endif()
math(EXPR LENGTH "${END}-${START}")
string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} SOURCE)
string(REPLACE "int vsl_process_layout(VslEngine *e," "int vsl_audition_process_layout(VslEngine *e,VslSourceAudition *observer," SOURCE "${SOURCE}")
string(REPLACE "int channels,float *stems)" "int channels,float *stems,float *views)" SOURCE "${SOURCE}")
string(REPLACE "vsl_scene_update(e,channels);" "vsl_scene_update(e,channels);vsl_audition_update(e,observer,channels);" SOURCE "${SOURCE}")
set(MARKER "vsl_scene_tick(e,o,spatial,channels);++e->frame;")
string(FIND "${SOURCE}" "${MARKER}" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen spatial observer hook missing")
endif()
string(REPLACE "${MARKER}" "vsl_scene_tick(e,o,spatial,channels);vsl_audition_tick(e,observer,o,spatial,views?views+i*VSL_AUDITION_MODES*channels:NULL,channels);++e->frame;" SOURCE "${SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/audition_engine.c" "#include \"audition.h\"\n#include \"core/modules.h\"\n#include \"core/character.h\"\n#include \"drive/drive.h\"\n#include \"spatial/scene.h\"\n#include <math.h>\n#include <string.h>\n${SOURCE}")
file(READ "${VSL_FROZEN}/session/take.c" SOURCE)
string(FIND "${SOURCE}" "unsigned vsl_take_process(" START)
if(START EQUAL -1)
  message(FATAL_ERROR "Frozen take process boundary missing")
endif()
string(SUBSTRING "${SOURCE}" ${START} -1 SOURCE)
string(REPLACE "unsigned vsl_take_process(VslTakePlayer *player,VslEngine *e," "unsigned vsl_audition_take_process(VslTakePlayer *player,VslEngine *e,VslSourceAudition *observer," SOURCE "${SOURCE}")
string(REPLACE "int channels,float *stems)" "int channels,float *stems,float *views)" SOURCE "${SOURCE}")
set(MARKER "vsl_process_layout(e,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL);")
string(FIND "${SOURCE}" "${MARKER}" HOOK)
if(HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen take dispatch hook missing")
endif()
string(REPLACE "${MARKER}" "vsl_audition_process_layout(e,observer,out+done*channels,n,channels,stems?stems+done*VSL_OBJECTS:NULL,views?views+done*VSL_AUDITION_MODES*channels:NULL);" SOURCE "${SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/audition_take.c" "#include \"audition.h\"\n#include <math.h>\n${SOURCE}")
add_library(source_audition STATIC audition.c "${CMAKE_CURRENT_BINARY_DIR}/audition_scene.c" "${CMAKE_CURRENT_BINARY_DIR}/audition_engine.c" "${CMAKE_CURRENT_BINARY_DIR}/audition_take.c")
target_include_directories(source_audition PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${VSL_FROZEN}" "${VSL_FROZEN}/core")
target_link_libraries(source_audition PUBLIC "${VSL_ROOT}/build-v016/Release/vsl_core.lib")
add_executable(source_audition_probe probe.c)
target_link_libraries(source_audition_probe PRIVATE source_audition)
add_executable(source_audition_checks checks.c)
target_link_libraries(source_audition_checks PRIVATE source_audition)
enable_testing()
add_test(NAME source_audition_contracts COMMAND source_audition_checks)
if(MSVC)
  foreach(target source_audition source_audition_probe source_audition_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
endif()
