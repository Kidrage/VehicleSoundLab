cmake_minimum_required(VERSION 3.21)
project(VslBandlimitedDna LANGUAGES C)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}/experiments/catalog_drive_v1" previous-system)
add_library(bandlimited_test_models STATIC test_models.c)
target_link_libraries(bandlimited_test_models PUBLIC dna_atlas)
add_executable(bandlimited_load_checks load_checks.c)
target_link_libraries(bandlimited_load_checks PRIVATE bandlimited_test_models)
add_executable(bandlimited_render_checks render_checks.c)
target_link_libraries(bandlimited_render_checks PRIVATE bandlimited_test_models)
add_executable(bandlimited_drive_checks drive_checks.c)
target_link_libraries(bandlimited_drive_checks PRIVATE bandlimited_test_models atlas_take)
add_executable(sound_dna_bandlimited_export probe.c "${VSL_ROOT}/src/io/wav.c" "${VSL_ROOT}/src/io/file.c")
target_include_directories(sound_dna_bandlimited_export PRIVATE "${VSL_ROOT}/src")
target_link_libraries(sound_dna_bandlimited_export PRIVATE dna_atlas)
add_executable(sound_dna_legacy_export "${VSL_ROOT}/experiments/sound_dna_export_v1/probe.c" "${VSL_ROOT}/src/io/wav.c" "${VSL_ROOT}/src/io/file.c")
target_include_directories(sound_dna_legacy_export PRIVATE "${VSL_ROOT}/src")
target_link_libraries(sound_dna_legacy_export PRIVATE sound_dna)
foreach(target bandlimited_test_models bandlimited_load_checks bandlimited_render_checks bandlimited_drive_checks sound_dna_bandlimited_export sound_dna_legacy_export)
  target_compile_features(${target} PRIVATE c_std_11)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  endif()
endforeach()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/bandlimited-check")
add_test(NAME bandlimited_profile_and_bank_transaction COMMAND bandlimited_load_checks "${CMAKE_BINARY_DIR}/bandlimited-check/model.txt")
add_test(NAME bandlimited_analytic_stereo_sweep_and_chunks COMMAND bandlimited_render_checks)
add_test(NAME bandlimited_high_anchor_same_driver_2_12 COMMAND bandlimited_drive_checks)
set_tests_properties(bandlimited_profile_and_bank_transaction bandlimited_analytic_stereo_sweep_and_chunks bandlimited_high_anchor_same_driver_2_12 PROPERTIES WORKING_DIRECTORY "${VSL_ROOT}")
