cmake_minimum_required(VERSION 3.21)
project(VslCharacterExperiment LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VSL_FROZEN "${VSL_ROOT}/baselines/v0.15.0/src")
# Offline object hook only. Keep every other byte of the frozen dispatcher.
file(READ "${VSL_FROZEN}/core/engine.c" VSL_ENGINE_TEXT)
set(VSL_HOOK_MARKER "        ir=vsl_ir_tick(e,.5*(o->sample[0]+o->sample[1]));")
string(REGEX MATCHALL "ir=vsl_ir_tick" VSL_MARKERS "${VSL_ENGINE_TEXT}")
list(LENGTH VSL_MARKERS VSL_MARKER_COUNT)
if(NOT VSL_MARKER_COUNT EQUAL 1)
  message(FATAL_ERROR "Frozen engine hook location is ambiguous")
endif()
string(FIND "${VSL_ENGINE_TEXT}" "${VSL_HOOK_MARKER}" VSL_MARKER_POSITION)
if(VSL_MARKER_POSITION EQUAL -1)
  message(FATAL_ERROR "Frozen engine hook location changed")
endif()
string(REPLACE "${VSL_HOOK_MARKER}" "        vsl_character_hook(e,o);\n${VSL_HOOK_MARKER}" VSL_ENGINE_TEXT "${VSL_ENGINE_TEXT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/engine_hook.c" "#include \"character.h\"\n${VSL_ENGINE_TEXT}")
add_library(character STATIC character.c pressure.c structure.c profile.c "${CMAKE_CURRENT_BINARY_DIR}/engine_hook.c")
target_include_directories(character PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${VSL_FROZEN}" "${VSL_FROZEN}/core")
target_link_libraries(character PUBLIC "${VSL_ROOT}/build-v015/Release/vsl_core.lib")
add_executable(character_probe probe.c)
target_link_libraries(character_probe PRIVATE character)
add_executable(character_checks checks.c)
target_link_libraries(character_checks PRIVATE character)
enable_testing()
add_test(NAME character_source_contracts COMMAND character_checks)
if(MSVC)
  target_compile_options(character PRIVATE /W4 /we4013 /utf-8)
  target_compile_options(character_probe PRIVATE /W4 /we4013 /utf-8)
  target_compile_options(character_checks PRIVATE /W4 /we4013 /utf-8)
endif()
