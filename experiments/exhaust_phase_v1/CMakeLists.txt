cmake_minimum_required(VERSION 3.21)
project(VslExhaustPhaseExperiment LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VSL_FROZEN "${VSL_ROOT}/baselines/v0.16.0/src")
# Generate only a copied process function with an explicit instance argument.
# The native core and its source files are never patched or recompiled here.
file(READ "${VSL_FROZEN}/core/engine.c" VSL_ENGINE)
string(FIND "${VSL_ENGINE}" "int vsl_process_layout(" VSL_START)
string(FIND "${VSL_ENGINE}" "void vsl_process(" VSL_END)
if(VSL_START EQUAL -1 OR VSL_END EQUAL -1 OR VSL_END LESS VSL_START)
  message(FATAL_ERROR "Frozen process boundaries changed")
endif()
math(EXPR VSL_LENGTH "${VSL_END}-${VSL_START}")
string(SUBSTRING "${VSL_ENGINE}" ${VSL_START} ${VSL_LENGTH} VSL_PROCESS)
set(VSL_MARKER "        ir=vsl_ir_tick(e,.5*(o->sample[0]+o->sample[1]));")
string(FIND "${VSL_PROCESS}" "${VSL_MARKER}" VSL_HOOK)
if(VSL_HOOK EQUAL -1)
  message(FATAL_ERROR "Frozen source-IR hook changed")
endif()
string(REPLACE "int vsl_process_layout(VslEngine *e," "int vsl_phase_process_layout(VslEngine *e,VslExhaustTransfer *transfer," VSL_PROCESS "${VSL_PROCESS}")
string(REPLACE "${VSL_MARKER}" "        vsl_phase_tick(transfer,e,o);\n${VSL_MARKER}" VSL_PROCESS "${VSL_PROCESS}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/phase_engine.c" "#include \"transfer.h\"\n#include \"core/modules.h\"\n#include \"core/character.h\"\n#include \"drive/drive.h\"\n#include \"spatial/scene.h\"\n#include \"spatial/propagation.h\"\n#include <math.h>\n#include <string.h>\n${VSL_PROCESS}")
file(READ "${VSL_FROZEN}/session/take.c" VSL_TAKE)
string(FIND "${VSL_TAKE}" "unsigned vsl_take_process(" VSL_START)
if(VSL_START EQUAL -1)
  message(FATAL_ERROR "Frozen take process boundary changed")
endif()
string(SUBSTRING "${VSL_TAKE}" ${VSL_START} -1 VSL_TAKE)
string(REPLACE "unsigned vsl_take_process(VslTakePlayer *player,VslEngine *e," "unsigned vsl_phase_take_process(VslTakePlayer *player,VslEngine *e,VslExhaustTransfer *transfer," VSL_TAKE "${VSL_TAKE}")
string(REPLACE "vsl_process_layout(e," "vsl_phase_process_layout(e,transfer," VSL_TAKE "${VSL_TAKE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/phase_take.c" "#include \"transfer.h\"\n#include <math.h>\n${VSL_TAKE}")
add_library(exhaust_phase STATIC allpass.c profile.c transfer.c "${CMAKE_CURRENT_BINARY_DIR}/phase_engine.c" "${CMAKE_CURRENT_BINARY_DIR}/phase_take.c")
target_include_directories(exhaust_phase PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${VSL_FROZEN}")
target_link_libraries(exhaust_phase PUBLIC "${VSL_ROOT}/build-v016/Release/vsl_core.lib")
add_executable(exhaust_phase_probe probe.c)
target_link_libraries(exhaust_phase_probe PRIVATE exhaust_phase)
add_executable(exhaust_phase_checks checks.c)
target_link_libraries(exhaust_phase_checks PRIVATE exhaust_phase)
enable_testing()
add_test(NAME causal_exhaust_phase_contracts COMMAND exhaust_phase_checks)
if(MSVC)
  foreach(target exhaust_phase exhaust_phase_probe exhaust_phase_checks)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8)
  endforeach()
endif()
