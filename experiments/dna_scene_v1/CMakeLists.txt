cmake_minimum_required(VERSION 3.21)
project(VslDnaSceneV1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}" legacy EXCLUDE_FROM_ALL)
add_subdirectory("${VSL_ROOT}/modules/sound_dna" sound-dna)
add_subdirectory("${VSL_ROOT}/modules/dna_scene" dna-scene)
add_executable(dna_scene_probe probe.c)
target_link_libraries(dna_scene_probe PRIVATE dna_scene)
add_executable(dna_scene_checks checks.c source_checks.c spatial_checks.c)
target_link_libraries(dna_scene_checks PRIVATE dna_scene)
if(MSVC)
  target_compile_options(dna_scene_probe PRIVATE /W4 /we4013 /utf-8)
  target_compile_options(dna_scene_checks PRIVATE /W4 /we4013 /utf-8)
endif()
add_test(NAME dna_scene_contract COMMAND dna_scene_checks
  "${VSL_ROOT}/voices/dna_bank_v1/V10 Exterior.model"
  "${VSL_ROOT}/voices/coherent_low_v1/measured_phase.model")
