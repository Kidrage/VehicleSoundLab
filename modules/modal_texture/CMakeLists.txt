add_library(modal_texture STATIC profile.c envelope.c voice.c)
target_include_directories(modal_texture PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(modal_texture PUBLIC modal_source)
add_library(modal_texture_io STATIC io.c "${CMAKE_CURRENT_SOURCE_DIR}/../../src/io/file.c")
target_include_directories(modal_texture_io PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
target_link_libraries(modal_texture_io PUBLIC modal_texture)
foreach(target modal_texture modal_texture_io)
  target_compile_features(${target} PUBLIC c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_libraries(${target} PUBLIC m)
  endif()
endforeach()
