add_library(spectral_texture STATIC profile.c voice.c)
target_include_directories(spectral_texture PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(spectral_texture PUBLIC partitioned_fir resonant_detail)
add_library(spectral_texture_io STATIC io.c "${CMAKE_CURRENT_SOURCE_DIR}/../../src/io/file.c")
target_include_directories(spectral_texture_io PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
target_link_libraries(spectral_texture_io PUBLIC spectral_texture)
foreach(target spectral_texture spectral_texture_io)
  target_compile_features(${target} PUBLIC c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endforeach()
