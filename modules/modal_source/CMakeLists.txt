add_library(modal_source STATIC profile.c voice.c)
target_include_directories(modal_source PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(modal_source_io STATIC io.c "${CMAKE_CURRENT_SOURCE_DIR}/../../src/io/file.c")
target_link_libraries(modal_source_io PUBLIC modal_source)
target_include_directories(modal_source_io PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
foreach(target modal_source modal_source_io)
  target_compile_features(${target} PUBLIC c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /we4013 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_libraries(${target} PUBLIC m)
  endif()
endforeach()
