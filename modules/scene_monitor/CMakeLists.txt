add_library(scene_monitor STATIC render.c)
target_include_directories(scene_monitor PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(scene_monitor PUBLIC dna_scene)
target_compile_features(scene_monitor PUBLIC c_std_11)
if(MSVC)
  target_compile_options(scene_monitor PRIVATE /W4 /we4013 /utf-8)
else()
  target_compile_options(scene_monitor PRIVATE -Wall -Wextra -Wpedantic)
endif()
