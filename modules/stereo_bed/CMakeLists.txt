add_library(stereo_bed STATIC route.c)
target_include_directories(stereo_bed PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(stereo_bed PUBLIC c_std_11)
if(MSVC)
  target_compile_options(stereo_bed PRIVATE /W4 /we4013 /utf-8)
else()
  target_compile_options(stereo_bed PRIVATE -Wall -Wextra -Wpedantic)
endif()
