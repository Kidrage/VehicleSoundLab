add_library(event_control STATIC scheduler.c policy.c configuration.c)
target_include_directories(event_control PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(event_control PUBLIC c_std_11)
if(MSVC)
  target_compile_options(event_control PRIVATE /W4 /we4013 /utf-8 /fp:strict)
else()
  target_compile_options(event_control PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(event_control PUBLIC m)
endif()
