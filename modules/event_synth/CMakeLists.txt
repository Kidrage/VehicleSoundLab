add_library(event_synth STATIC profile.c oscillator.c excitation.c coloration.c envelope.c contour.c voice.c pool.c)
target_include_directories(event_synth PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")

endif()
