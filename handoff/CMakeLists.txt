cmake_minimum_required(VERSION 3.21)
project(VehicleSoundLabHandoff VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(VSL_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
enable_testing()
add_subdirectory("${VSL_ROOT}" drive)
add_subdirectory("${VSL_ROOT}/modules/sound_dna" sound-dna)
add_subdirectory("${VSL_ROOT}/modules/low_rumble" low-rumble)

function(handoff_output target)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endfunction()
foreach(target vsl_render vsl_take_render vsl_spatial_probe)
  handoff_output(${target})
endforeach()

if(WIN32)
  handoff_output(VehicleSoundLab)
  add_executable(VslSystem WIN32 launcher/main.c)
  target_compile_options(VslSystem PRIVATE /W4 /we4013 /utf-8)
  target_link_libraries(VslSystem PRIVATE shell32)
  handoff_output(VslSystem)

  function(handoff_monitor target revision)
    set(m "${VSL_ROOT}/experiments/sustain_monitor_v${revision}")
    add_library(${target}_render STATIC "${m}/render.c")
    target_link_libraries(${target}_render PUBLIC sound_dna)
    if(revision EQUAL 4)
      target_link_libraries(${target}_render PUBLIC low_rumble)
    endif()
    add_executable(${target} WIN32 "${m}/ui.c" "${m}/audio.c" "${m}/smoke.c" "${m}/bank.c")
    target_link_libraries(${target} PRIVATE ${target}_render vsl_core comctl32 shell32 winmm avrt comdlg32)
    add_executable(${target}_probe "${m}/probe.c")
    target_link_libraries(${target}_probe PRIVATE ${target}_render vsl_core)
    add_executable(${target}_checks "${m}/checks.c")
    target_link_libraries(${target}_checks PRIVATE ${target}_render)
    foreach(t ${target} ${target}_render ${target}_probe ${target}_checks)
      target_compile_options(${t} PRIVATE /W4 /we4013 /utf-8)
    endforeach()
    handoff_output(${target})
    handoff_output(${target}_probe)
    set(dna "${VSL_ROOT}/voices/dna_bank_v1/V10 Exterior.model")
    if(revision EQUAL 4)
      add_test(NAME handoff_low_rumble_contract COMMAND ${target}_checks "${dna}" "${VSL_ROOT}/voices/low_rumble_v2/A Exterior.model")
    else()
      add_test(NAME handoff_sound_dna_contract COMMAND ${target}_checks "${dna}")
    endif()
  endfunction()
  handoff_monitor(VslSoundDna 3)
  handoff_monitor(VslLowRumble 4)
endif()
